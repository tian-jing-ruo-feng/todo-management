# 数据库规范

本文档定义了 Todo Management 项目的数据库设计规范，基于 Dexie（IndexedDB 封装）和 DexieCloud。

## 数据库架构

### Dexie 配置

项目使用 Dexie 作为 IndexedDB 的封装，配置 DexieCloud 实现云端同步。

```typescript
import Dexie from 'dexie'
import { dexieCloud } from 'dexie-cloud-addon'

export class ToDoDb extends Dexie {
  tasks!: Dexie.Table<Task>
  statuses!: Dexie.Table<Status>
  priorities!: Dexie.Table<Priority>
  groups!: Dexie.Table<Group>

  constructor() {
    super('ToDoDB', {
      addons: [dexieCloud],
    })

    this.version(1).stores({
      tasks: 'id, status, priority, createTime, updateTime, isTop, isRemoved, sort',
      statuses: 'id',
      priorities: 'id',
      groups: 'id',
    })

    this.cloud.configure({
      databaseUrl: import.meta.env.VITE_DATABASE_URL,
      requireAuth: true,
      customLoginGui: true,
      // disableEagerSync: true, // 可选：禁用自动同步
    })

    this.tasks.mapToClass(Task)
    this.statuses.mapToClass(Status)
    this.priorities.mapToClass(Priority)
    this.groups.mapToClass(Group)
  }
}

const db = new ToDoDb()
export default db
```

### 数据库版本管理

使用版本号管理数据库结构变更。

```typescript
this.version(1).stores({
  tasks: 'id, name, status, priority',
  statuses: 'id',
  priorities: 'id',
  groups: 'id',
})

// 版本升级示例
this.version(2).stores({
  tasks: 'id, name, status, priority, sort, isTop', // 添加新索引
  statuses: 'id',
  priorities: 'id',
  groups: 'id',
}).upgrade(tx => {
  // 数据迁移逻辑
  return tx.table('tasks').toCollection().modify(task => {
    task.sort = 0
    task.isTop = false
  })
})
```

#### Dexie Entity 类定义

```typescript
// src/db/task.ts
import { Entity } from 'dexie'
import type { ToDoDb } from './db'

export class Task extends Entity<ToDoDb> {
  id!: string
  realmId!: string
  owner!: string
  name!: string
  content?: string
  status?: string
  isRemoved?: boolean
  group?: string[]
  createTime?: string
  updateTime?: string
  expectStartTime?: string
  expectEndTime?: string
  priority!: string
  isTop?: boolean
  sort?: number

  // 业务方法
  isSharable() {
    return this.realmId === getTiedRealmId(this.id)
  }

  async makeSharable() {
    const realmId = getTiedRealmId(this.id)
    const db = this.db
    await db.transaction('rw', [db.tasks, db.realms], () => {
      db.realms.upsert(realmId, {
        name: this.name,
        represents: '一个任务设置',
      })
      db.tasks.update(this.id!, { realmId: realmId })
    })
    return realmId
  }

  async makePrivate() {
    const tiedRealmId = getTiedRealmId(this.id)
    const db = this.db
    await db.transaction('rw', [db.tasks, db.members, db.realms], async () => {
      await db.tasks.where({ id: this.id, realmId: tiedRealmId }).modify({
        realmId: db.cloud.currentUserId,
        owner: db.cloud.currentUserId,
      })
      await db.members.where('realmId').equals(tiedRealmId).delete()
      await db.realms.delete(tiedRealmId)
    })
  }

  async deleteTask() {
    const { db } = this
    await db.transaction('rw', [db.tasks, db.realms], async () => {
      db.tasks.delete(this.id)
      const tiedRealmId = getTiedRealmId(this.id)
      await db.realms.delete(tiedRealmId)
    })
  }
}
```

## 数据操作规范

### Repository 模式

使用 Repository 模式封装数据库操作，提高代码的可维护性和可测试性。

```typescript
// src/utils/repositories/StatusRepository.ts
import { db } from '@/db/db'

export class StatusRepository {
  /**
   * 获取所有状态
   */
  async getAll(): Promise<Status[]> {
    return await db.statuses.toArray()
  }

  /**
   * 根据ID获取状态
   */
  async getById(id: string): Promise<Status | undefined> {
    return await db.statuses.get(id)
  }

  /**
   * 添加状态
   */
  async add(status: Status): Promise<string> {
    return await db.statuses.add(status)
  }

  /**
   * 更新状态
   */
  async update(id: string, changes: Partial<Status>): Promise<number> {
    return await db.statuses.update(id, changes)
  }

  /**
   * 删除状态
   */
  async delete(id: string): Promise<void> {
    await db.statuses.delete(id)
  }

  /**
   * 批量添加
   */
  async bulkAdd(statuses: Status[]): Promise<string[]> {
    return await db.statuses.bulkAdd(statuses)
  }
}

export const statusRepository = new StatusRepository()
```

### CRUD 操作

#### 创建（Create）

```typescript
/**
 * 添加或更新任务
 * @param task 任务对象，如果包含id则为更新，否则为添加
 * @returns Promise<string> 返回任务ID
 */
export const saveTask = async (task: Task): Promise<string> => {
  // 如果任务存在id，则更新；否则添加
  if (task.id && (await db.tasks.get(task.id))) {
    await db.tasks.update(task.id, task)
    return task.id
  } else {
    const newTaskData = {
      ...task,
      id: `task_${Date.now()}_${Math.random().toString(36).substring(2, 11)}`,
      createTime: new Date().toISOString(),
      updateTime: new Date().toISOString(),
    }
    await db.tasks.add(newTaskData)
    return newTaskData.id
  }
}
```

## 数据加密

对于敏感数据，使用加密存储。

```typescript
// src/utils/crypto.ts
import CryptoJS from 'crypto-js'

/**
 * 加密任务字段
 */
export function encryptTaskFields(task: Task, key: string): Task {
  const encrypted = {
    ...task,
    name: CryptoJS.AES.encrypt(task.name, key).toString(),
  }

  if (task.content) {
    encrypted.content = CryptoJS.AES.encrypt(task.content, key).toString()
  }

  return encrypted
}

/**
 * 解密任务字段
 */
export function decryptTaskFields(task: Task, key: string): Task {
  const decrypted = {
    ...task,
    name: CryptoJS.AES.decrypt(task.name, key).toString(CryptoJS.enc.Utf8),
  }

  if (task.content) {
    decrypted.content = CryptoJS.AES.decrypt(
      task.content,
      key
    ).toString(CryptoJS.enc.Utf8)
  }

  return decrypted
}

/**
 * 批量加密
 */
export function encryptTasks(tasks: Task[], key: string): Task[] {
  return tasks.map((task) => encryptTaskFields(task, key))
}

/**
 * 批量解密
 */
export function decryptTasks(tasks: Task[], key: string): Task[] {
  return tasks.map((task) => decryptTaskFields(task, key))
}
```

使用加密：

```typescript
const ENCRYPTION_KEY = import.meta.env.VITE_ENCRYPTION_KEY || ''

// 保存时加密
export const saveTask = async (task: Task): Promise<string> => {
  const encryptedTask = ENCRYPTION_KEY
```

## 数据验证

### 输入验证

```typescript
/**
 * 验证任务数据
 */
export function validateTask(task: Partial<Task>): boolean {
  if (!task.name || task.name.trim().length === 0) {
    throw new Error('任务名称不能为空')
  }

  if (task.name.length > 200) {
    throw new Error('任务名称不能超过200个字符')
  }

  if (!task.status) {
    throw new Error('任务状态不能为空')
  }

  if (!task.priority) {
    throw new Error('任务优先级不能为空')
  }

  return true
}

/**
 * 保存任务（带验证）
 */
export const saveTaskWithValidation = async (task: Task): Promise<string> => {
  validateTask(task)
  return await saveTask(task)
}
```

```typescript
export const importData = async (jsonData: string): Promise<void> => {
  const data = JSON.parse(jsonData)

  await db.transaction('rw', db.tasks, db.statuses, db.priorities, db.groups, async () => {
```

```typescript
/**
 * 分批加载数据
 */
export const loadTasksInBatches = async (
  batchSize: number = 100
): Promise<Task[]> => {
  let offset = 0
  let allTasks: Task[] = []
  let hasMore = true

  while (hasMore) {
    const tasks = await db.tasks.offset(offset).limit(batchSize).toArray()
    if (tasks.length === 0) {
      hasMore = false
    } else {
      allTasks = allTasks.concat(tasks)
      offset += batchSize
    }
  }

  return allTasks
}
```

## 最佳实践

### 1. 使用 Repository 模式

将数据库操作封装在 Repository 类中，避免直接在组件中操作数据库。

```typescript
// ✅ 好的实践
const tasks = await taskRepository.getAll()

// ❌ 不好的实践
const tasks = await db.tasks.toArray()
```

### 2. 使用软删除

不直接删除数据，而是标记为已删除，便于数据恢复。

```typescript
// ✅ 好的实践
export const deleteTask = async (taskId: string): Promise<void> => {
  await db.tasks.update(taskId, { isRemoved: true })
}

// ❌ 不好的实践
export const deleteTask = async (taskId: string): Promise<void> => {
  await db.tasks.delete(taskId)
}
```

### 3. 添加时间戳

所有数据都应包含创建时间和更新时间。

```typescript
const newTask = {
  ...taskData,
  createTime: new Date().toISOString(),
  updateTime: new Date().toISOString(),
}
```

### 4. 使用事务

对于涉及多个表的操作，使用事务确保数据一致性。

```typescript
await db.transaction('rw', db.tasks, db.logs, async () => {
  await db.tasks.add(task)

```

### 5. 处理错误


### 6. 使用索引

为经常查询的字段创建索引，提高查询性能。

```typescript
this.version(1).stores({
  tasks: 'id, status, priority, createTime', // 为常用字段创建索引

```

## 总结

遵循以上数据库规范，可以确保：

- **数据一致性**：使用事务和验证机制
- **查询性能**：合理使用索引和批量操作
- **数据安全**：实现加密和软删除机制
- **可维护性**：使用 Repository 模式封装数据操作
- **可扩展性**：灵活的数据模型设计
